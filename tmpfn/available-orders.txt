import { createClient } from "jsr:@supabase/supabase-js@2";

const EARTH_RADIUS_KM = 6371;

// CORS: без этого браузер режет preflight (OPTIONS) — у мастеров «Failed to fetch».
const CORS_HEADERS = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Headers": "authorization, apikey, content-type, x-client-info",
  "Access-Control-Allow-Methods": "POST, OPTIONS",
};

function distanceKm(a, b) {
  const dLat = ((b.lat - a.lat) * Math.PI) / 180;
  const dLng = ((b.lng - a.lng) * Math.PI) / 180;
  const lat1 = (a.lat * Math.PI) / 180;
  const lat2 = (b.lat * Math.PI) / 180;
  const h = Math.sin(dLat / 2) ** 2 + Math.cos(lat1) * Math.cos(lat2) * Math.sin(dLng / 2) ** 2;
  const c = 2 * Math.atan2(Math.sqrt(h), Math.sqrt(1 - h));
  return EARTH_RADIUS_KM * c;
}

function json(obj, status = 200) {
  return new Response(JSON.stringify(obj), { status, headers: { "Content-Type": "application/json", ...CORS_HEADERS } });
}

Deno.serve(async (req) => {
  if (req.method === "OPTIONS") return new Response(null, { status: 204, headers: CORS_HEADERS });
  try {
    if (req.method !== "POST") return json({ ok: false, error: "method_not_allowed" }, 405);
    const body = await req.json().catch(() => ({}));
    const workerId = body.workerId;
    const sessionToken = body.sessionToken;
    if (!workerId || !sessionToken) return json({ ok: false, error: "auth_required" }, 400);

    const supabase = createClient(Deno.env.get("SUPABASE_URL"), Deno.env.get("SUPABASE_SERVICE_ROLE_KEY"));

    const { data: session, error: sessionErr } = await supabase.from("user_sessions").select("token").eq("user_id", workerId).maybeSingle();
    if (sessionErr) throw sessionErr;
    if (!session || session.token !== sessionToken) return json({ ok: false, error: "unauthorized" }, 401);

    const { data: worker, error: workerErr } = await supabase
      .from("users")
      .select("id, lat, lng, available_as_worker, blocked, search_radius_km, lat2, lng2")
      .eq("id", workerId)
      .maybeSingle();
    if (workerErr) throw workerErr;
    if (!worker || worker.blocked || !worker.available_as_worker) return json({ ok: true, orders: [] });

    const { data: skillRows, error: skillErr } = await supabase.from("skills").select("category_id, note").eq("user_id", workerId);
    if (skillErr) throw skillErr;

    // Навыки с ограничением из-за брака (view worker_skill_stats): заказы этих категорий мастер не видит, пока админ не снимет ограничение.
    const { data: restrRows } = await supabase.from("worker_skill_stats").select("category_id").eq("worker_id", workerId).eq("restricted", true);
    const restricted = new Set((restrRows || []).map((r) => r.category_id));
    const categoryIds = Array.from(new Set((skillRows || []).map((s) => s.category_id))).filter((c) => !restricted.has(c));
    if (!categoryIds.length) return json({ ok: true, orders: [] });
    // Сложные заказы (orders.complex) — только по навыкам с уровнем 4–5★ или PRO.
    const qualified = new Set((skillRows || []).filter((s) => /^(PRO::|LVL[45]::)/.test(String(s.note || ""))).map((s) => s.category_id));

    const { data: orderRows, error: orderErr } = await supabase
      .from("orders")
      .select(
        "id, order_number, customer_id, title, category_id, description, desired_outcome, scope, has_voice_transcript, exact_lat, exact_lng, district, house_number_public, desired_date, time_window, urgency, estimate, status, warnings, is_precedent, created_at, device_model, diagnosis_log, estimated_minutes, worker_note, parts_already_have, parts_maybe_needed, parts_needed, formatted_brief, is_test, timeline, search_radius_km, complex",
      )
      .eq("status", "searching")
      // «Другое» видят все мастера, кроме тех, у кого из-за брака ограничена категория «Другое».
      .or(restricted.has("other") ? `category_id.in.(${categoryIds.join(",")})` : `category_id.in.(${categoryIds.join(",")}),category_id.eq.other`);
    if (orderErr) throw orderErr;

    const radiusKm = Number(worker.search_radius_km ?? 3) || 3;
    const workerLoc = { lat: worker.lat, lng: worker.lng };
    const hasLoc2 = worker.lat2 != null && worker.lng2 != null && (worker.lat2 !== 0 || worker.lng2 !== 0);
    const workerLoc2 = hasLoc2 ? { lat: worker.lat2, lng: worker.lng2 } : null;

    const { data: blockRows } = await supabase.from("master_customer_blocks").select("customer_id").eq("worker_id", workerId);
    const blockedCustomers = new Set((blockRows || []).map((b) => b.customer_id));
    const custIds = [...new Set((orderRows || []).map((r) => r.customer_id).filter(Boolean))];
    const custMap = {};
    if (custIds.length) {
      const { data: custRows } = await supabase.from("users").select("id, display_name, avatar_url").in("id", custIds);
      for (const c of custRows || []) custMap[c.id] = c;
    }
    const orders = (orderRows || []).filter((r) => !(r.complex && !qualified.has(r.category_id || "other")) && !(Array.isArray(r.timeline) && r.timeline.some((t) => t && t.declined && t.workerId === workerId)) && !blockedCustomers.has(r.customer_id) && (r.customer_id !== workerId || r.is_test) && (!r.is_test || workerId === "id1"))
      .map((row) => ({
        row,
        dKm: Math.min(distanceKm(workerLoc, { lat: row.exact_lat, lng: row.exact_lng }), workerLoc2 ? distanceKm(workerLoc2, { lat: row.exact_lat, lng: row.exact_lng }) : Infinity),
      }))
      // Радиус заказа сайт расширяет ступенями (expand_order_radii), если рядом мало мастеров: тогда мастер видит и более дальние заказы.
      .filter((x) => x.dKm <= Math.max(radiusKm, Number(x.row.search_radius_km) || 0))
      .sort((x, y) => x.dKm - y.dKm)
      .map(({ row, dKm }) => ({
        id: row.id,
        isTest: Boolean(row.is_test),
        customerName: (custMap[row.customer_id] && custMap[row.customer_id].display_name) || "",
        customerAvatarUrl: (custMap[row.customer_id] && custMap[row.customer_id].avatar_url) || "",
        orderNumber: row.order_number ?? null,
        title: row.title,
        categoryId: row.category_id,
        description: row.description || "",
        desiredOutcome: row.desired_outcome || "",
        scope: row.scope || {},
        hasVoiceTranscript: Boolean(row.has_voice_transcript),
        district: String(row.district || "").replace(/[,;]?\s*(?:кв\.?|квартира|кв-ра|оф\.?|офис|комн\.?|комната)\s*№?\s*\d+[а-яa-z]?/gi, "").replace(/[,;]?\s*(?:подъезд|под\.?|этаж|эт\.?|домофон)\s*№?\s*\d+[а-яa-z]?/gi, "").replace(/[,;\s]+$/g, "").trim(),
        houseNumberPublic: row.house_number_public || "",
        desiredDate: row.desired_date || "",
        timeWindow: row.time_window || "",
        urgency: row.urgency || "normal",
        estimate: row.estimate || { min: 0, max: 0, factors: [] },
        status: row.status,
        warnings: row.warnings || [],
        isPrecedent: Boolean(row.is_precedent),
        createdAt: row.created_at,
        distanceKm: Math.round(dKm * 10) / 10,
        far: dKm > radiusKm,
        complex: Boolean(row.complex),
        // Модель устройства и лента диагностики видны мастеру ещё до того, как он взял заказ.
        deviceModel: row.device_model || "",
        diagnosisLog: row.diagnosis_log || [],
        estimatedMinutes: row.estimated_minutes ?? null,
        workerNote: row.worker_note || "",
        partsAlreadyHave: row.parts_already_have || "",
        partsMaybeNeeded: row.parts_maybe_needed || "",
        partsNeeded: row.parts_needed || "",
        formattedBrief: row.formatted_brief || null,
        lat: Math.round(row.exact_lat * 10000) / 10000,
        lng: Math.round(row.exact_lng * 10000) / 10000,
      }));

    return json({ ok: true, orders });
  } catch (e) {
    console.error(e);
    return json({ ok: false, error: String((e && e.message) || e) }, 500);
  }
});
